current_staff_id = ContextVar("current_staff_id", default=None)
current_role = ContextVar("current_role", default=None)
current_hospital_id = ContextVar("current_hospital_id", default=None)

def set_identity(staff_id, role, hospital_id, chat_session_id=None):
    current_staff_id.set(staff_id)
    current_role.set(role)
    current_hospital_id.set(hospital_id)
